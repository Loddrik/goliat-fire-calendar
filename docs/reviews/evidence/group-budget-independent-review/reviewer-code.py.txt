def risk(ds,a):return math.fsum(math.fsum(float(w)/math.fsum(map(float,weights))*abs(float(v)-float(t)) for v,w in zip(values,weights)) for (values,weights),t in zip(ds,a))
def kkt(ds,a):
 lower,upper=-1.,1.
 for (v,w),t in zip(ds,a):
  den=math.fsum(map(float,w));eps=1e-10*np.maximum(1.,np.maximum(np.abs(v),abs(t)));l=2*math.fsum(float(z)/den for z in w[v<t-eps])-1;u=2*math.fsum(float(z)/den for z in w[v<=t+eps])-1
  if t>1e-10*max(1.,abs(t)):lower=max(lower,l)
  upper=min(upper,u)
 return max(0.,lower-upper)
