def oldcdf(b,cut,g,s,parent=False):
 h=b['hurdle'];p=h['parents'][key(g,s)];prob=float(p['p']);z=list(p['positive']);w=[1.]*len(z)
 l=h['locals'].get(key(cut,g,s),{'n':0,'m':0,'positive':[]})
 if h['all_zero']:return np.array([0.]),np.array([1.])
 if not parent and l['n']:
  oldp=prob;prob=(l['m']+10*oldp)/(l['n']+10)
  if l['m']:
   z=list(l['positive'])+z;w=[1.]*l['m']+[10*oldp/len(p['positive'])]*len(p['positive'])
 if not prob:return np.array([0.]),np.array([1.])
 return np.array([0.]+z),np.array([1-prob]+[prob*x/math.fsum(w) for x in w])
def mix(st,prior):
 if not st['n']:return prior
 v,w=prior;n=st['n'];emp=[0.]*(n-st['m'])+st['positive']
 return np.array(emp+list(v)),np.array([1/(n+10)]*n+[float(x)*10/(n+10) for x in w])
def cdf(model,b,cut,g,a,d):
 s=int(a>0);pool=model['pools'][key(g,s)];meta={}

  v,w=oldcdf(b,cut,g,s);meta={'fallback':'original_full_CDF_no_SOURCE_D_variation','parent_n':pool['n'],'local_n':None}
 else:
  st=pool['classes'][str(d)];raw=mix(st,oldcdf(b,cut,g,s,True));local=model['locals'].get(key(cut,g,s,d),{'n':0,'m':0,'positive':[]}) if cut in model['source_cuts'] else {'n':0,'m':0,'positive':[]}
  v,w=mix(local,raw);meta={'fallback':'contextual_local' if local['n'] else 'contextual_parent','parent_n':st['n'],'parent_m':st['m'],'local_n':local['n'],'local_m':local['m'],'parent_borrowed_mass':10/(st['n']+10),'local_borrowed_mass':10/(local['n']+10)}
 return (np.where(v>0,np.maximum(1.,v*(1+a*b['regional_ratio'][g])),0.),w),meta

def canonical(ds):
 v,w=ds;ans={}

def equalcdf(a,b):

def median(ds):
 items=canonical(ds);threshold=math.fsum(w for _,w in items)/2;acc=0.
 for v,w in items:
  acc+=w
  if acc>=threshold:return v
 return items[-1][0]
def risk(ds,a):return math.fsum(math.fsum(float(w)/math.fsum(map(float,ww))*abs(float(v)-float(t)) for v,w in zip(vv,ww)) for (vv,ww),t in zip(ds,a))
def kkt(ds,a):
 low,high=-1.,1.
 for (v,w),t in zip(ds,a):
  den=math.fsum(map(float,w));eps=1e-10*np.maximum(1.,np.maximum(abs(v),abs(t)));lo=2*math.fsum(float(z)/den for z in w[v<t-eps])-1;hi=2*math.fsum(float(z)/den for z in w[v<=t+eps])-1
  if t>1e-10*max(1.,abs(t)):low=max(low,lo)
  high=min(high,hi)
 return max(0.,low-high)

 return out
